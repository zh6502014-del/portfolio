⚙️ 設定 Supabase 連線

在開始使用後台之前，請先完成 Supabase 設定。這是免費的雲端資料庫，用來儲存您的作品集內容與圖片。

1
前往 supabase.com 免費註冊並建立新專案
2
在 Project Settings → API 找到 Project URL 和 anon public key
3
在 SQL Editor 執行下方的建表語法
4
在 Storage 建立名為 portfolio-images 的 bucket，設定為 Public
📋 SQL 建表語法（點擊展開） 
-- 個人資料
CREATE TABLE IF NOT EXISTS profile (
  id SERIAL PRIMARY KEY,
  name TEXT,
  nav_name TEXT,
  tagline TEXT,
  description TEXT,
  email TEXT,
  phone TEXT,
  status TEXT DEFAULT '目前在職',
  updated_at TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE profile
ADD COLUMN IF NOT EXISTS nav_name TEXT;

-- 技能
CREATE TABLE IF NOT EXISTS skills (
  id SERIAL PRIMARY KEY,
  title TEXT NOT NULL,
  icon TEXT DEFAULT '🎨',
  tags TEXT[] DEFAULT '{}',
  sort_order INT DEFAULT 0
);

-- 工具
CREATE TABLE IF NOT EXISTS tools (
  id SERIAL PRIMARY KEY,
  name TEXT NOT NULL,
  description TEXT,
  sort_order INT DEFAULT 0
);

-- 專案
CREATE TABLE IF NOT EXISTS projects (
  id SERIAL PRIMARY KEY,
  title TEXT NOT NULL,
  time_range TEXT,
  tags TEXT[] DEFAULT '{}',
  description TEXT,
  content TEXT,
  sections JSONB DEFAULT '[]'::jsonb,
  role TEXT,
  members TEXT,
  goals TEXT[] DEFAULT '{}',
  challenges TEXT[] DEFAULT '{}',
  methods TEXT[] DEFAULT '{}',
  work_items TEXT[] DEFAULT '{}',
  products TEXT[] DEFAULT '{}',
  insight TEXT,
  images TEXT[] DEFAULT '{}',
  sort_order INT DEFAULT 0,
  published BOOLEAN DEFAULT TRUE,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE projects
ADD COLUMN IF NOT EXISTS content TEXT,
ADD COLUMN IF NOT EXISTS sections JSONB DEFAULT '[]'::jsonb;

-- 隨筆
CREATE TABLE IF NOT EXISTS notes (
  id SERIAL PRIMARY KEY,
  title TEXT NOT NULL,
  note_date DATE,
  note_type TEXT,
  content TEXT,
  sort_order INT DEFAULT 0,
  published BOOLEAN DEFAULT TRUE
);

-- 導覽列
CREATE TABLE IF NOT EXISTS nav_items (
  id SERIAL PRIMARY KEY,
  label TEXT NOT NULL,
  href TEXT NOT NULL,
  sort_order INT DEFAULT 0
);
INSERT INTO nav_items (label, href, sort_order)
SELECT label, href, sort_order
FROM (VALUES
  ('關於', '#about', 1),
  ('作品集', '#projects', 2),
  ('隨筆', '#notes', 3)
) AS v(label, href, sort_order)
WHERE NOT EXISTS (SELECT 1 FROM nav_items);
DELETE FROM nav_items a
USING nav_items b
WHERE a.href = b.href
  AND a.id > b.id;

CREATE UNIQUE INDEX IF NOT EXISTS nav_items_href_unique
ON nav_items (href);

-- RLS (開放匿名讀取，只有登入者可寫)
ALTER TABLE profile ENABLE ROW LEVEL SECURITY;
ALTER TABLE skills ENABLE ROW LEVEL SECURITY;
ALTER TABLE tools ENABLE ROW LEVEL SECURITY;
ALTER TABLE projects ENABLE ROW LEVEL SECURITY;
ALTER TABLE notes ENABLE ROW LEVEL SECURITY;
ALTER TABLE nav_items ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "public read" ON profile;
DROP POLICY IF EXISTS "auth write" ON profile;
DROP POLICY IF EXISTS "public read" ON skills;
DROP POLICY IF EXISTS "auth write" ON skills;
DROP POLICY IF EXISTS "public read" ON tools;
DROP POLICY IF EXISTS "auth write" ON tools;
DROP POLICY IF EXISTS "public read" ON projects;
DROP POLICY IF EXISTS "auth write" ON projects;
DROP POLICY IF EXISTS "public read" ON notes;
DROP POLICY IF EXISTS "auth write" ON notes;
DROP POLICY IF EXISTS "public read" ON nav_items;
DROP POLICY IF EXISTS "auth write" ON nav_items;

CREATE POLICY "public read" ON profile FOR SELECT USING (true);
CREATE POLICY "auth write" ON profile FOR ALL USING (auth.role() = 'authenticated');
CREATE POLICY "public read" ON skills FOR SELECT USING (true);
CREATE POLICY "auth write" ON skills FOR ALL USING (auth.role() = 'authenticated');
CREATE POLICY "public read" ON tools FOR SELECT USING (true);
CREATE POLICY "auth write" ON tools FOR ALL USING (auth.role() = 'authenticated');
CREATE POLICY "public read" ON projects FOR SELECT USING (true);
CREATE POLICY "auth write" ON projects FOR ALL USING (auth.role() = 'authenticated');
CREATE POLICY "public read" ON notes FOR SELECT USING (true);
CREATE POLICY "auth write" ON notes FOR ALL USING (auth.role() = 'authenticated');
CREATE POLICY "public read" ON nav_items FOR SELECT USING (true);
CREATE POLICY "auth write" ON nav_items FOR ALL USING (auth.role() = 'authenticated');

-- 前台區塊顯示開關
CREATE TABLE IF NOT EXISTS section_settings (
  key TEXT PRIMARY KEY,
  visible BOOLEAN NOT NULL DEFAULT TRUE,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE section_settings ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "public read" ON section_settings;
DROP POLICY IF EXISTS "auth write" ON section_settings;
CREATE POLICY "public read" ON section_settings FOR SELECT USING (true);
CREATE POLICY "auth write" ON section_settings FOR ALL USING (auth.role() = 'authenticated');

-- 頁尾項目（個人資料頁 → 頁尾；footer_text 存 JSON 陣列，footer_link_label 為舊版欄位）
ALTER TABLE profile
ADD COLUMN IF NOT EXISTS footer_text TEXT,
ADD COLUMN IF NOT EXISTS footer_link_label TEXT;
